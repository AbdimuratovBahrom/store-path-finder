 document.addEventListener("DOMContentLoaded", () => {
  const blockSelect = document.getElementById("blockSelect");
  const rowSelect = document.getElementById("rowSelect");
  const shopSelect = document.getElementById("shopSelect");
  const resultDiv = document.getElementById("result");

  blockSelect.addEventListener("change", async () => {
    const block = blockSelect.value;
    rowSelect.innerHTML = "<option>Загрузка...</option>";
    shopSelect.innerHTML = "<option>Выберите магазин</option>";
    rowSelect.disabled = true;
    shopSelect.disabled = true;

    if (!block) return;

    const res = await fetch(`/get_rows/${block}`);
    const data = await res.json();

    if (data.type === "shops") {
      shopSelect.innerHTML = "<option>Выберите магазин</option>";
      data.items.forEach((shop) => {
        const opt = document.createElement("option");
        opt.value = shop;
        opt.textContent = shop;
        shopSelect.appendChild(opt);
      });
      shopSelect.disabled = false;
    } else {
      rowSelect.innerHTML = "<option>Выберите ряд</option>";
      data.items.forEach((row) => {
        const opt = document.createElement("option");
        opt.value = row;
        opt.textContent = row;
        rowSelect.appendChild(opt);
      });
      rowSelect.disabled = false;
    }
  });

  rowSelect.addEventListener("change", async () => {
    const block = blockSelect.value;
    const row = rowSelect.value;
    shopSelect.innerHTML = "<option>Загрузка...</option>";
    shopSelect.disabled = true;

    const res = await fetch(`/get_stores/${block}/${row}`);
    const data = await res.json();
    shopSelect.innerHTML = "<option>Выберите магазин</option>";
    data.items.forEach((shop) => {
      const opt = document.createElement("option");
      opt.value = shop;
      opt.textContent = shop;
      shopSelect.appendChild(opt);
    });
    shopSelect.disabled = false;
  });

  document.getElementById("searchBtn").addEventListener("click", async () => {
    const block = blockSelect.value;
    const row = rowSelect.value || "None";
    const shop = shopSelect.value;
    if (!block || !shop) {
      resultDiv.textContent = "Выберите блок и магазин.";
      return;
    }
    const res = await fetch(`/get_path/${block}/${row}/${shop}`);
    const data = await res.json();
    resultDiv.innerHTML = data.path ? `<b>${data.path}</b>` : data.error;
  });
});





import sqlite3
import re
from flask import Flask, render_template, request, jsonify, session, redirect, url_for
from flask_babel import Babel, gettext as _, get_locale

app = Flask(__name__)
app.secret_key = "supersecretkey"

# === Языки ===
LANGUAGES = {
    "ru": "Русский",
    "uz_Latn": "O‘zbekcha (Lotin)",
    "uz_Cyrl": "Ўзбекча (Кирилл)"
}

def select_locale():
    return session.get("lang", "ru")

babel = Babel(app, locale_selector=select_locale)

@app.context_processor
def inject_get_locale():
    return dict(get_locale=get_locale)

# === Подключение к БД ===
def get_db_connection():
    conn = sqlite3.connect("shops.db")
    conn.row_factory = sqlite3.Row
    return conn

# === Сортировка ===
def numeric_key(s):
    if s is None:
        return (10**9, "")
    m = re.search(r"(\d+)", str(s))
    if m:
        return (int(m.group(1)), str(s))
    return (10**9, str(s))

# === Фильтр для спецобъектов ===
def is_specific_shop(name):
    if not name:
        return False
    s = str(name).strip().lower()
    s = re.sub(r"^маг[- ]*", "", s)
    if re.fullmatch(r"\d+", s):
        return False
    if re.fullmatch(r"\d+[a-zа-яё]+[0-9]*", s):
        return False
    if s == "005":
        return False
    return True

# === Маршруты ===
@app.route("/")
def index():
    conn = get_db_connection()
    blocks = [r["block"] for r in conn.execute("SELECT DISTINCT block FROM shops").fetchall()]
    conn.close()

    if "Специфические объекты" not in blocks:
        blocks.append("Специфические объекты")

    return render_template("index.html", blocks=sorted(blocks))

@app.route("/set_language/<lang>")
def set_language(lang):
    if lang in LANGUAGES:
        session["lang"] = lang
    return redirect(url_for("index"))

# === Получение рядов ===
@app.route("/get_rows/<block>")
def get_rows(block):
    conn = get_db_connection()

    # Гипермаркет — сразу магазины
    if block == "Гипермаркет":
        stores = conn.execute("SELECT DISTINCT shop FROM shops WHERE block = ?", (block,)).fetchall()
        conn.close()
        items = sorted({s["shop"] for s in stores}, key=numeric_key)
        return jsonify({"type": "shops", "items": items})

    # Специфические объекты — сразу магазины
    if block == "Специфические объекты":
        stores = conn.execute(
            "SELECT DISTINCT shop FROM shops WHERE row='Спец' OR block IN ('1-блок','2-блок','3-блок')"
        ).fetchall()
        conn.close()
        items = sorted({s["shop"] for s in stores if is_specific_shop(s["shop"])}, key=str)
        return jsonify({"type": "shops", "items": items})

    # Обычные блоки — вернуть ряды
    rows = conn.execute(
        "SELECT DISTINCT row FROM shops WHERE block = ? AND row IS NOT NULL", (block,)
    ).fetchall()
    conn.close()

    def row_key(x):
        if not x:
            return (2, "")
        if re.match(r"^\d+$", str(x)):
            return (0, int(x))
        if re.match(r"^[A-Za-zА-Яа-яЁё]+$", str(x)):
            return (1, str(x))
        return (2, str(x))

    items = sorted({r["row"] for r in rows}, key=row_key)
    return jsonify({"type": "rows", "items": items})

# === Получение магазинов ===
@app.route("/get_stores/<block>/<row>")
def get_stores(block, row):
    conn = get_db_connection()

    if block == "Специфические объекты":
        stores = conn.execute(
            "SELECT DISTINCT shop FROM shops WHERE row='Спец' OR block IN ('1-блок','2-блок','3-блок')"
        ).fetchall()
        conn.close()
        items = sorted({s["shop"] for s in stores if is_specific_shop(s["shop"])}, key=str)
        return jsonify({"items": items})

    if block == "Гипермаркет":
        stores = conn.execute(
            "SELECT DISTINCT shop FROM shops WHERE block = ?", (block,)
        ).fetchall()
        conn.close()
        items = sorted({s["shop"] for s in stores}, key=numeric_key)
        return jsonify({"items": items})

    if row in ("None", "", "None "):
        stores = conn.execute(
            "SELECT DISTINCT shop FROM shops WHERE block = ? AND row IS NULL",
            (block,)
        ).fetchall()
    else:
        stores = conn.execute(
            "SELECT DISTINCT shop FROM shops WHERE block = ? AND row = ?",
            (block, row)
        ).fetchall()
    conn.close()

    items = sorted({s["shop"] for s in stores}, key=numeric_key)
    return jsonify({"items": items})

# === Получение пути ===
@app.route("/get_path/<block>/<row>/<shop>")
def get_path(block, row, shop):
    conn = get_db_connection()
    add_mag = not (block in ["Гипермаркет", "Специфические объекты"])

    if block == "Специфические объекты":
        results = conn.execute(
            "SELECT DISTINCT path FROM shops WHERE shop = ?", (shop,)
        ).fetchall()
    elif row in ("None", "", "None "):
        results = conn.execute(
            "SELECT DISTINCT path FROM shops WHERE block = ? AND row IS NULL AND shop = ?",
            (block, shop)
        ).fetchall()
    else:
        results = conn.execute(
            "SELECT DISTINCT path FROM shops WHERE block = ? AND row = ? AND shop = ?",
            (block, row, shop)
        ).fetchall()

    conn.close()

    if not results:
        return jsonify({"error": _("Путь не найден")})

    paths = []
    for r in results:
        path = r["path"]
        if add_mag and re.fullmatch(r"\d+", shop.strip()):
            path = f"{path} > Маг-{shop}"
        paths.append(path)

    paths = list(dict.fromkeys(paths))
    return jsonify({"path": " | ".join(paths)})

# === Поиск ===
@app.route("/search")
def search():
    keyword = request.args.get("keyword", "").strip()
    conn = get_db_connection()
    results = conn.execute(
        "SELECT * FROM shops WHERE shop LIKE ? OR block LIKE ? OR row LIKE ? OR path LIKE ?",
        (f"%{keyword}%", f"%{keyword}%", f"%{keyword}%", f"%{keyword}%")
    ).fetchall()
    conn.close()
    formatted = [
        f"Block: {r['block']} | Row: {r['row']} | Shop: {r['shop']} | Path: {r['path']}"
        for r in results
    ]
    formatted = list(dict.fromkeys(formatted))
    return jsonify({"results": formatted} if formatted else {"error": _("Ничего не найдено")})

if __name__ == "__main__":
    app.run(debug=True)
